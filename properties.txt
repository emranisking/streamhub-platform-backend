# =========================================================
# StreamHub Platform - Main Configuration
# =========================================================

spring.application.name=streamhub-platform
server.port=${SERVER_PORT:8080}
server.servlet.context-path=/

# ---------------------------------------------------------
# Datasource (PostgreSQL)
# ---------------------------------------------------------
spring.datasource.url=${DB_URL:jdbc:postgresql://localhost:5432/streamhub}
spring.datasource.username=${DB_USERNAME:streamhub}
spring.datasource.password=${DB_PASSWORD:streamhub}
spring.datasource.driver-class-name=org.postgresql.Driver

# ---------------------------------------------------------
# JPA / Hibernate
# ---------------------------------------------------------
# Flyway owns the schema. Hibernate must never auto-generate DDL in production.
spring.jpa.hibernate.ddl-auto=validate
spring.jpa.open-in-view=false
spring.jpa.properties.hibernate.jdbc.time_zone=UTC
spring.jpa.properties.hibernate.format_sql=false
spring.jpa.show-sql=false

# ---------------------------------------------------------
# Flyway migrations
# ---------------------------------------------------------
spring.flyway.enabled=true
spring.flyway.locations=classpath:db/migration
spring.flyway.baseline-on-migrate=true

# ---------------------------------------------------------
# Redis
# ---------------------------------------------------------
spring.data.redis.host=${REDIS_HOST:localhost}
spring.data.redis.port=${REDIS_PORT:6379}
spring.data.redis.password=${REDIS_PASSWORD:}
spring.data.redis.timeout=3000ms

# Global cache TTL (seconds) - used across every @Cacheable region and manual cache writes
app.redis.cache-ttl-seconds=${CACHE_TTL_SECONDS:3600}

# ---------------------------------------------------------
# CORS - single global frontend URL used across the whole app
# Comma separated list is supported for multiple environments.
# ---------------------------------------------------------
app.cors.frontend-url=${FRONTEND_URL:http://localhost:5173}

# ---------------------------------------------------------
# JWT / Security
# ---------------------------------------------------------
app.security.jwt.secret=${JWT_SECRET:this-is-a-development-only-secret-please-change-in-production-min-256-bits}
app.security.jwt.expiration-ms=${JWT_EXPIRATION_MS:604800000}
# 604800000 ms = 7 days

# ---------------------------------------------------------
# Pagination / URL security
# Salt used to sign pagination cursors so page/limit values
# handed back to the client cannot be tampered with.
# ---------------------------------------------------------
app.pagination.salt=${PAGINATION_SALT:change-this-pagination-salt-in-production}
app.pagination.default-page=0
app.pagination.default-limit=20
app.pagination.max-limit=100

# ---------------------------------------------------------
# Playback / guest watch limiting
# ---------------------------------------------------------
app.playback.free-limit=${PLAYBACK_FREE_LIMIT:2}
app.playback.guest-ttl-days=${PLAYBACK_GUEST_TTL_DAYS:30}

# ---------------------------------------------------------
# Video storage / conversion
# ---------------------------------------------------------
app.media.source-directory=${MEDIA_SOURCE_DIR:./media/incoming}
app.media.hls-output-directory=${MEDIA_HLS_DIR:./media/hls}
app.media.public-route-prefix=/media
app.media.ffmpeg-path=${FFMPEG_PATH:ffmpeg}
app.media.scan-on-startup=true

# ---------------------------------------------------------
# Actuator
# ---------------------------------------------------------
management.endpoints.web.exposure.include=health,info
management.endpoint.health.show-details=when-authorized

# ---------------------------------------------------------
# Logging
# ---------------------------------------------------------
logging.level.root=INFO
logging.level.com.streamhub.platform=INFO
logging.level.com.streamhub.platform.common.aop=INFO
logging.pattern.console=%d{yyyy-MM-dd HH:mm:ss} %-5level [%thread] %logger{36} - %msg%n

# ---------------------------------------------------------
# springdoc
# ---------------------------------------------------------
springdoc.api-docs.path=/api-docs
springdoc.swagger-ui.path=/swagger-ui.html
